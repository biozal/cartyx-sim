````ts
/** A heading-bounded slice of one markdown file, the unit that gets embedded and retrieved. */
export interface Chunk {
  chunkId: string;
  source: string;
  title: string;
  headings: string[];
  text: string;
}

/** A section longer than this splits again on paragraph boundaries. */
export const CHUNK_MAX_CHARS = 4000;
/** A section with less body text than this is a bare heading or a nav stub, and is dropped. */
export const CHUNK_MIN_CHARS = 80;

const TITLE_SEPARATOR = ' › ';

export function slugify(text: string): string {
  return (
    text
      .toLowerCase()
      .normalize('NFKD')
      // Keep letters and digits from any script; everything else becomes a separator.
      .replace(/[^\p{L}\p{N}]+/gu, '-')
      .replace(/^-+|-+$/g, '') || 'section'
  );
}

interface Section {
  headings: string[];
  lines: string[];
}

/**
 * Splits markdown into sections at every ATX heading, tracking the heading path. A fenced code
 * block's `#` lines are body text, not headings.
 */
function toSections(markdown: string): Section[] {
  const sections: Section[] = [{ headings: [], lines: [] }];
  const path: { level: number; text: string }[] = [];
  let fence: string | undefined;
  for (const line of markdown.split(/\r?\n/)) {
    const fenceMatch = /^\s*(```+|~~~+)/.exec(line);
    if (fenceMatch) {
      const marker = fenceMatch[1]!;
      if (fence === undefined) fence = marker[0];
      else if (marker[0] === fence) fence = undefined;
      sections.at(-1)!.lines.push(line);
      continue;
    }
    const heading = fence === undefined ? /^(#{1,6})\s+(.*\S)\s*$/.exec(line) : null;
    if (!heading) {
      sections.at(-1)!.lines.push(line);
      continue;
    }
    const level = heading[1]!.length;
    const text = heading[2]!;
    while (path.length > 0 && path.at(-1)!.level >= level) path.pop();
    path.push({ level, text });
    sections.push({ headings: path.map((entry) => entry.text), lines: [] });
  }
  return sections;
}

/** Greedily packs paragraphs into parts no longer than {@link CHUNK_MAX_CHARS}. */
function splitLong(text: string): string[] {
  if (text.length <= CHUNK_MAX_CHARS) return [text];
  const parts: string[] = [];
  let current = '';
  for (const paragraph of text.split(/\n{2,}/)) {
    const candidate = current === '' ? paragraph : `${current}\n\n${paragraph}`;
    if (candidate.length <= CHUNK_MAX_CHARS || current === '') {
      current = candidate;
      // A single paragraph over the cap becomes its own oversized part rather than being cut
      // mid-sentence: retrieval quality matters more here than a hard byte ceiling.
      if (current.length > CHUNK_MAX_CHARS) {
        parts.push(current);
        current = '';
      }
      continue;
    }
    parts.push(current);
    current = paragraph;
  }
  if (current !== '') parts.push(current);
  return parts;
}

export function chunkMarkdown(source: string, markdown: string): Chunk[] {
  const chunks: Chunk[] = [];
  for (const section of toSections(markdown)) {
    const text = section.lines.join('\n').trim();
    if (text.length < CHUNK_MIN_CHARS) continue;
    const title = section.headings.length === 0 ? source : section.headings.join(TITLE_SEPARATOR);
    const slugPath = section.headings.map(slugify).join('/');
    const parts = splitLong(text);
    parts.forEach((part, index) => {
      const suffix = parts.length === 1 ? '' : `~${index + 1}`;
      chunks.push({
        chunkId: `${source}#${slugPath}${suffix}`,
        source,
        title,
        headings: [...section.headings],
        text: part,
      });
    });
  }
  return chunks;
}
````

import type { Chunk } from './chunk';
import { slugify } from './chunk';

/** One entity from the Kanka export mirror. */
export interface KankaEntity {
  entityId: string;
  kankaId: number;
  entityType: string;
  entityName: string;
}

export interface EntityIndex {
  /** Normalized name -> the one entity with that name, or 'ambiguous' when several share it. */
  byName: ReadonlyMap<string, KankaEntity | 'ambiguous'>;
  /** Display names that resolved to more than one entity, for `sim index` to report. */
  ambiguous: readonly string[];
}

/** What a chunk records about the entity it describes. */
export interface ChunkEntity {
  entityId: string;
  entityName: string;
  entityType: string;
  kankaId?: number;
}

/**
 * Titles a lore writer uses in prose but Kanka usually omits. Stripped only when something follows,
 * so an NPC actually named "Captain" keeps their name.
 */
const HONORIFICS = [
  'professor',
  'prof',
  'headmistress',
  'headmaster',
  'dean',
  'doctor',
  'dr',
  'mr',
  'mrs',
  'ms',
  'miss',
  'lord',
  'lady',
  'captain',
  'sir',
  'dame',
];

export function normalizeName(name: string): string {
  const base = name
    .toLowerCase()
    .normalize('NFKD')
    .replace(/[^\p{L}\p{N}\s]+/gu, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  const [first, ...rest] = base.split(' ');
  if (first !== undefined && rest.length > 0 && HONORIFICS.includes(first)) return rest.join(' ');
  return base;
}

/** `kanka-export/markdown/<type>/<id>-<id>.md` -> the entity it holds. */
const KANKA_PATH = /^kanka-export\/markdown\/([a-z]+)\/(\d+)-\d+\.md$/;

/** `characters` -> `character`; `families` -> `family`. */
function singular(folder: string): string {
  if (folder.endsWith('ies')) return `${folder.slice(0, -3)}y`;
  return folder.endsWith('s') ? folder.slice(0, -1) : folder;
}

export function parseKankaFile(path: string, markdown: string): KankaEntity | undefined {
  const match = KANKA_PATH.exec(path.replaceAll('\\', '/'));
  if (!match) return undefined;
  const heading = /^#\s+(.*\S)\s*$/m.exec(markdown);
  if (!heading) return undefined;
  const kankaId = Number(match[2]);
  if (!Number.isSafeInteger(kankaId) || kankaId <= 0) return undefined;
  return {
    entityId: `kanka-${kankaId}`,
    kankaId,
    entityType: singular(match[1]!),
    entityName: heading[1]!,
  };
}

export function buildEntityIndex(files: readonly { path: string; text: string }[]): EntityIndex {
  const byName = new Map<string, KankaEntity | 'ambiguous'>();
  const ambiguous = new Map<string, string>();
  for (const file of files) {
    const entity = parseKankaFile(file.path, file.text);
    if (!entity) continue;
    const key = normalizeName(entity.entityName);
    if (key === '') continue;
    const existing = byName.get(key);
    if (existing === undefined) {
      byName.set(key, entity);
      continue;
    }
    // A second file for the same Kanka id is the same entity, not a collision.
    if (existing !== 'ambiguous' && existing.kankaId === entity.kankaId) continue;
    byName.set(key, 'ambiguous');
    ambiguous.set(key, entity.entityName);
  }
  return { byName, ambiguous: [...ambiguous.values()].sort() };
}

function pathEntity(chunk: Chunk): ChunkEntity {
  const slugPath = chunk.headings.map(slugify).join('/');
  return {
    entityId: `file:${chunk.source}#${slugPath}`,
    entityName: chunk.headings.at(-1) ?? chunk.source,
    entityType: 'document',
  };
}

/**
 * Walks a chunk's heading path deepest-first and takes the first heading that uniquely names a
 * Kanka entity, so an NPC's section resolves to the NPC and a "Campus" section under a school
 * resolves to the school. An ambiguous name is never guessed: it falls through to the path id.
 */
export function resolveEntity(chunk: Chunk, index: EntityIndex): ChunkEntity {
  for (let depth = chunk.headings.length - 1; depth >= 0; depth--) {
    const match = index.byName.get(normalizeName(chunk.headings[depth]!));
    // An unknown heading ("Campus") keeps walking outward to its parent. An *ambiguous* one stops
    // the walk: the heading does name an entity, we just cannot tell which, and resolving it to
    // the enclosing location instead would be a quiet miscount rather than an honest fallback.
    if (match === 'ambiguous') return pathEntity(chunk);
    if (match === undefined) continue;
    return {
      entityId: match.entityId,
      entityName: match.entityName,
      entityType: match.entityType,
      kankaId: match.kankaId,
    };
  }
  return pathEntity(chunk);
}

import { createHash } from 'node:crypto';
import { readdir, readFile } from 'node:fs/promises';
import { join, relative, sep } from 'node:path';

/** One markdown file from the lore repository. */
export interface LoreFile {
  path: string;
  text: string;
}

// Only these two. Skipping a directory that holds no markdown today (images/, kanka-export/_raw/)
// buys a little walk speed and costs a lore file added there later, silently. Spec §3.
const SKIP_DIRS = new Set(['node_modules', '.git']);

/**
 * Everything markdown is lore, including the Kanka mirror. The only markdown left out is the
 * export's generated navigation, which is a list of links rather than anything a DM could use.
 */
export function isIndexable(relativePath: string): boolean {
  const path = relativePath.replaceAll('\\', '/');
  if (!path.endsWith('.md')) return false;
  if (path.split('/').some((part) => SKIP_DIRS.has(part))) return false;
  const name = path.split('/').at(-1)!;
  if (path.startsWith('kanka-export/') && (name === '_index.md' || name === 'index.md')) {
    return false;
  }
  return true;
}

export async function readLoreFiles(repoPath: string): Promise<LoreFile[]> {
  const files: LoreFile[] = [];
  const walk = async (dir: string): Promise<void> => {
    const entries = await readdir(dir, { withFileTypes: true });
    for (const entry of entries) {
      const full = join(dir, entry.name);
      const rel = relative(repoPath, full).split(sep).join('/');
      if (entry.isDirectory()) {
        if (SKIP_DIRS.has(entry.name)) continue;
        await walk(full);
        continue;
      }
      if (!entry.isFile() || !isIndexable(rel)) continue;
      files.push({ path: rel, text: await readFile(full, 'utf8') });
    }
  };
  await walk(repoPath);
  files.sort((a, b) => (a.path < b.path ? -1 : a.path > b.path ? 1 : 0));
  return files;
}

/**
 * A digest over the paths and bytes actually ingested. This, not the git commit, is what decides
 * whether an index is stale: a commit touching no lore file must not cost a rebuild, and an
 * uncommitted edit must.
 */
export function contentDigest(files: readonly LoreFile[]): string {
  const lines = files
    .map((file) => `${file.path}\0${createHash('sha256').update(file.text).digest('hex')}`)
    .sort();
  const digest = createHash('sha256').update(lines.join('\n')).digest('hex');
  return `sha256:${digest}`;
}
```

import { execFile } from 'node:child_process';
import { promisify } from 'node:util';

const run = promisify(execFile);

async function git(repoPath: string, args: string[]): Promise<string | undefined> {
  try {
    const { stdout } = await run('git', args, { cwd: repoPath });
    return stdout.trim();
  } catch {
    return undefined;
  }
}

/**
 * The lore version recorded in `session_start` and compared on resume: the commit sha when the
 * tree is clean, and `<sha>-dirty-<digest8>` when it is not, so an uncommitted lore edit is a
 * different version and resuming into it is correctly refused.
 */
export async function loreVersion(repoPath: string, digest: string): Promise<string> {
  const hex = digest.replace(/^sha256:/, '');
  const head = await git(repoPath, ['rev-parse', 'HEAD']);
  if (head === undefined || !/^[0-9a-f]{40}$/.test(head)) return `nogit-${hex.slice(0, 16)}`;
  const status = await git(repoPath, ['status', '--porcelain']);
  if (status === '') return head;
  // A *failed* status check is not a clean tree. We do not know whether content differs, so say
  // so rather than stamping a bare sha — the digest still fingerprints the content, so this
  // string compares correctly either way: same content resumes, changed content is refused.
  if (status === undefined) return `${head}-unknown-${hex.slice(0, 8)}`;
  return `${head}-dirty-${hex.slice(0, 8)}`;
}

import type { ChunkEntity } from './entities';

/** A chunk as it sits in the index: its text, where it came from, and the entity it describes. */
export interface StoredChunk extends ChunkEntity {
  chunkId: string;
  source: string;
  title: string;
  text: string;
}

export interface ScoredChunk extends StoredChunk {
  /** Cosine similarity in [0, 1], matching the sense of the engine's `loreThreshold`. */
  score: number;
}

export interface VectorStore {
  search(vector: Float32Array, limit: number): Promise<ScoredChunk[]>;
  /**
   * Reads one chunk by id. `sim audit` needs it because the event log records chunk ids and
   * scores but never the chunk text, so a contradiction check has to fetch the text back.
   */
  byChunkId(chunkId: string): Promise<StoredChunk | undefined>;
  close(): Promise<void>;
}

export function cosineSimilarity(a: Float32Array, b: Float32Array): number {
  let dot = 0;
  let normA = 0;
  let normB = 0;
  const length = Math.min(a.length, b.length);
  for (let index = 0; index < length; index++) {
    const left = a[index]!;
    const right = b[index]!;
    dot += left * right;
    normA += left * left;
    normB += right * right;
  }
  if (normA === 0 || normB === 0) return 0;
  return dot / (Math.sqrt(normA) * Math.sqrt(normB));
}

/** The store the tests use: no LanceDB, no native addon, no I/O. */
export class MemoryStore implements VectorStore {
  constructor(private readonly rows: readonly { chunk: StoredChunk; vector: Float32Array }[]) {}

  async search(vector: Float32Array, limit: number): Promise<ScoredChunk[]> {
    return this.rows
      .map((row) => ({ ...row.chunk, score: cosineSimilarity(vector, row.vector) }))
      .sort((a, b) => b.score - a.score)
      .slice(0, limit);
  }

  async byChunkId(chunkId: string): Promise<StoredChunk | undefined> {
    return this.rows.find((row) => row.chunk.chunkId === chunkId)?.chunk;
  }

  async close(): Promise<void> {}
}

import { createOpenAICompatible } from '@ai-sdk/openai-compatible';
import type { ModelTarget } from '@cartyx-sim/models';
import { embed, embedMany } from 'ai';
import { createHash } from 'node:crypto';

/**
 * nomic-embed-text was trained with task prefixes, and retrieval quality drops sharply without
 * them. They are recorded in the index metadata, so changing one marks the index stale.
 */
export const DOCUMENT_PREFIX = 'search_document: ';
export const QUERY_PREFIX = 'search_query: ';

/** Chunks per embeddings request while indexing. */
export const EMBED_BATCH = 32;

export interface Embedder {
  embedDocuments(texts: readonly string[]): Promise<Float32Array[]>;
  embedQuery(text: string): Promise<Float32Array>;
  readonly model: string;
}

const PROVIDER_NAME = 'cartyx-sim-lore';

export class OpenAICompatibleEmbedder implements Embedder {
  readonly model: string;
  private readonly timeoutMs: number;
  private readonly target: ModelTarget;

  constructor(target: ModelTarget, options: { timeoutMs?: number } = {}) {
    this.target = target;
    this.model = target.model;
    this.timeoutMs = options.timeoutMs ?? 120_000;
  }

  private embeddingModel() {
    const provider = createOpenAICompatible({
      name: PROVIDER_NAME,
      baseURL: this.target.endpoint.baseURL,
      apiKey: this.target.endpoint.apiKey,
    });
    return provider.textEmbeddingModel(this.target.model);
  }

  async embedDocuments(texts: readonly string[]): Promise<Float32Array[]> {
    const { embeddings } = await embedMany({
      model: this.embeddingModel(),
      values: texts.map((text) => `${DOCUMENT_PREFIX}${text}`),
      // The caller owns retries and pausing, as it does for chat seats.
      maxRetries: 0,
      abortSignal: AbortSignal.timeout(this.timeoutMs),
    });
    return embeddings.map((vector) => Float32Array.from(vector));
  }

  async embedQuery(text: string): Promise<Float32Array> {
    const { embedding } = await embed({
      model: this.embeddingModel(),
      value: `${QUERY_PREFIX}${text}`,
      maxRetries: 0,
      abortSignal: AbortSignal.timeout(this.timeoutMs),
    });
    return Float32Array.from(embedding);
  }
}

/**
 * A deterministic hash-based embedder for tests. It has no semantic meaning — identical text gives
 * an identical vector and different text a different one, which is all the wiring tests need.
 */
export class FakeEmbedder implements Embedder {
  readonly model = 'fake-embedder';

  constructor(private readonly dims = 768) {}

  private vector(text: string): Float32Array {
    const out = new Float32Array(this.dims);
    for (let index = 0; index < this.dims; index++) {
      const digest = createHash('sha256').update(`${index}:${text}`).digest();
      // Map the first two bytes to [-1, 1] so vectors spread across the space.
      out[index] = (digest.readUInt16BE(0) / 0xffff) * 2 - 1;
    }
    return out;
  }

  async embedDocuments(texts: readonly string[]): Promise<Float32Array[]> {
    return texts.map((text) => this.vector(text));
  }

  async embedQuery(text: string): Promise<Float32Array> {
    return this.vector(text);
  }
}

import { chunkMarkdown } from './chunk';

import type { ScoredChunk, StoredChunk, VectorStore } from './store';

/** Bump when chunking or entity-resolution rules change, so old chunks are never reused. */
export const CHUNKER_VERSION = 1;

export interface IndexMeta {
  lorePath: string;
  loreCommit: string;
  contentDigest: string;
  embedModel: string;
  dims: number;
  documentPrefix: string;
  queryPrefix: string;
  chunkerVersion: number;
  files: number;
  chunks: number;
  /**
   * Entity names that matched two or more wiki entities and so fell back to a path id. Persisted
   * because they are a work item for the lore repo: printed once after a build and then lost, an
   * operator who missed the notice cannot get it back without --force. Additive — `isStale` must
   * NOT compare it, and a meta.json written before this field reads as `?? []`.
   */
  ambiguous: string[];
  builtAt: string;
}

/** The fields that decide staleness; `loreCommit` deliberately is not one of them. */
export type IndexIdentity = Omit<IndexMeta, 'loreCommit' | 'files' | 'chunks' | 'builtAt'>;

export interface BuildOptions {
  lorePath: string;
  indexDir: string;
  embedder: Embedder;
  log?: (line: string) => void;
}

export interface BuildResult {
  meta: IndexMeta;
  ambiguous: readonly string[];
}

const META_FILE = 'meta.json';

export async function readMeta(indexDir: string): Promise<IndexMeta | undefined> {
  try {
    return JSON.parse(await readFile(join(indexDir, META_FILE), 'utf8')) as IndexMeta;
  } catch {
    return undefined;
  }
}

export function isStale(meta: IndexMeta | undefined, current: IndexIdentity): boolean {
  if (!meta) return true;
  return (
    meta.lorePath !== current.lorePath ||
    meta.contentDigest !== current.contentDigest ||
    meta.embedModel !== current.embedModel ||
    meta.dims !== current.dims ||
    meta.documentPrefix !== current.documentPrefix ||
    meta.queryPrefix !== current.queryPrefix ||
    meta.chunkerVersion !== current.chunkerVersion
  );
}

export async function buildIndex(options: BuildOptions): Promise<BuildResult> {
  const log = options.log ?? (() => {});
  const files = await readLoreFiles(options.lorePath);
  log(`Read ${files.length} markdown files from ${options.lorePath}`);

  const entities = buildEntityIndex(files);
  const chunks: StoredChunk[] = [];
  for (const file of files) {
    for (const chunk of chunkMarkdown(file.path, file.text)) {
      chunks.push({
        chunkId: chunk.chunkId,
        source: chunk.source,
        title: chunk.title,
        text: chunk.text,
        ...resolveEntity(chunk, entities),
      });
    }
  }
  log(`Chunked into ${chunks.length} chunks across ${entities.byName.size} known entities`);

  const rows: { chunk: StoredChunk; vector: Float32Array }[] = [];
  for (let start = 0; start < chunks.length; start += EMBED_BATCH) {
    const batch = chunks.slice(start, start + EMBED_BATCH);
    // The heading path travels with the body so a short section keeps its context.
    const vectors = await options.embedder.embedDocuments(
      batch.map((chunk) => `${chunk.title}\n\n${chunk.text}`)
    );
    batch.forEach((chunk, index) => rows.push({ chunk, vector: vectors[index]! }));
    log(`Embedded ${Math.min(start + EMBED_BATCH, chunks.length)}/${chunks.length}`);
  }

  const digest = contentDigest(files);
  const meta: IndexMeta = {
    lorePath: options.lorePath,
    loreCommit: await loreVersion(options.lorePath, digest),
    contentDigest: digest,
    embedModel: options.embedder.model,
    dims: rows[0]?.vector.length ?? 0,
    documentPrefix: DOCUMENT_PREFIX,
    queryPrefix: QUERY_PREFIX,
    chunkerVersion: CHUNKER_VERSION,
    files: files.length,
    chunks: chunks.length,
    builtAt: new Date().toISOString(),
  };

  // Build beside the target and rename, so an interrupted index leaves the previous one intact.
  const staging = `${options.indexDir}.tmp-${process.pid}`;
  await rm(staging, { recursive: true, force: true });
  await mkdir(staging, { recursive: true });
  await LanceStore.write(staging, rows);
  // meta.json last: a half-written index must never look valid.
  await writeFile(join(staging, META_FILE), `${JSON.stringify(meta, null, 2)}\n`);
  await mkdir(dirname(options.indexDir), { recursive: true });
  await rm(options.indexDir, { recursive: true, force: true });
  await rename(staging, options.indexDir);

  return { meta, ambiguous: entities.ambiguous };
}

/** Wraps a store and an embedder as the engine's `LoreIndex`. */
export class LoreIndexImpl implements LoreIndex {
  constructor(
    private readonly store: VectorStore,
    private readonly embedder: Embedder
  ) {}

  async search(query: string, limit: number): Promise<LoreHit[]> {
    const vector = await this.embedder.embedQuery(query);
    // Exactly `limit`, as the interface promises. The over-fetching that feeds the entity
    // collapse happens in `lookup_lore`, which asks for a larger limit; doing it here too would
    // multiply the two factors together.
    const hits = await this.store.search(vector, limit);
    return hits.map(toLoreHit);
  }

  async close(): Promise<void> {
    await this.store.close();
  }
}

function toLoreHit(hit: ScoredChunk): LoreHit {
  return {
    chunkId: hit.chunkId,
    source: hit.source,
    title: hit.title,
    text: hit.text,
    score: hit.score,
    entityId: hit.entityId,
    entityName: hit.entityName,
    entityType: hit.entityType,
    ...(hit.kankaId === undefined ? {} : { kankaId: hit.kankaId }),
  };
}

export async function openIndex(indexDir: string, embedder: Embedder): Promise<LoreIndexImpl> {
  return new LoreIndexImpl(await LanceStore.open(indexDir), embedder);
}

import { writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import * as lancedb from '@lancedb/lancedb';
import type { ScoredChunk, StoredChunk, VectorStore } from './store';

const TABLE = 'chunks';

interface Row {
  vector: number[];
  chunkId: string;
  source: string;
  title: string;
  text: string;
  entityId: string;
  entityName: string;
  entityType: string;
  kankaId: number | null;
}

function toStored(row: Row): StoredChunk {
  return {
    chunkId: row.chunkId,
    source: row.source,
    title: row.title,
    text: row.text,
    entityId: row.entityId,
    entityName: row.entityName,
    entityType: row.entityType,
    ...(row.kankaId === null ? {} : { kankaId: row.kankaId }),
  };
}

export class LanceStore implements VectorStore {
  private constructor(
    private readonly connection: lancedb.Connection,
    private readonly table: lancedb.Table
  ) {}

  static async open(indexDir: string): Promise<LanceStore> {
    const connection = await lancedb.connect(indexDir);
    const table = await connection.openTable(TABLE);
    return new LanceStore(connection, table);
  }

  static async write(
    indexDir: string,
    rows: readonly { chunk: StoredChunk; vector: Float32Array }[]
  ): Promise<void> {
    const records: Row[] = rows.map(({ chunk, vector }) => ({
      vector: Array.from(vector),
      chunkId: chunk.chunkId,
      source: chunk.source,
      title: chunk.title,
      text: chunk.text,
      entityId: chunk.entityId,
      entityName: chunk.entityName,
      entityType: chunk.entityType,
      kankaId: chunk.kankaId ?? null,
    }));
    const connection = await lancedb.connect(indexDir);
    await connection.createTable(TABLE, records, { mode: 'overwrite' });
    // A plain JSON mirror of the rows, for tests and for reading an index without LanceDB.
    await writeFile(
      join(indexDir, 'chunks.json'),
      `${JSON.stringify(rows.map(({ chunk, vector }) => ({ chunk, vector: Array.from(vector) })))}\n`
    );
  }

  async search(vector: Float32Array, limit: number): Promise<ScoredChunk[]> {
    const rows = (await this.table
      .search(Array.from(vector))
      .distanceType('cosine')
      .limit(limit)
      .toArray()) as (Row & { _distance: number })[];
    return rows.map((row) => ({
      ...toStored(row),
      // LanceDB reports cosine *distance*; the engine's threshold is a similarity.
      score: 1 - row._distance,
    }));
  }

  async byChunkId(chunkId: string): Promise<StoredChunk | undefined> {
    const rows = (await this.table
      .query()
      // Lance's `where` predicate is SQL, where double quotes name a *column*, not a string —
      // JSON.stringify's double-quoted output would silently match against a (nonexistent) column
      // named after the id instead of comparing the value. Single-quoted string literals are the
      // correct form here, and doubling an embedded single quote escapes it.
      .where(`chunkId = '${chunkId.replaceAll("'", "''")}'`)
      .limit(1)
      .toArray()) as Row[];
    return rows[0] === undefined ? undefined : toStored(rows[0]);
  }

  async close(): Promise<void> {
    this.connection.close();
  }
}

      hits: ranked.map(({ chunkId, source, score, entityId, entityName }) => ({
        chunkId,
        source,
        score,
        entityId,
        entityName,
      })),
      used: [...usedIds],
    });
    if (used.length === 0) {
      return {
        result:
          'No lore found. If the story needs this detail, invent something consistent with the setting and call record_invention.',
      };
    }
    return {
      result: used
        .map((hit) => `[${hit.chunkId}] ${hit.title} (${hit.source})\n${hit.text}`)
        .join('\n\n'),
    };
  },
```

  /**
   * Attaches each invention to the narration it supports. Called once when the turn finishes,
   * before the director appends it: the DM looks lore up, finds a gap, records what it invented,
   * and only then narrates, so the link runs forward and falls back to the last narration before.
   */
  linkInventions(): void {
    for (const [index, event] of this.emitted.entries()) {
      if (event.type !== 'lore_invention' || event.narrationSeq !== undefined) continue;
      const after = this.emitted
        .slice(index + 1)
        .find((candidate) => candidate.type === 'narration');
      const before = this.emitted
        .slice(0, index)
        .findLast((candidate) => candidate.type === 'narration');
      const narration = after ?? before;
      if (narration) this.emitted[index] = { ...event, narrationSeq: narration.seq };
    }
  }

  OpenAICompatibleEmbedder,
  QUERY_PREFIX,

  type Embedder,
  type IndexIdentity,
  type IndexMeta,

import { EMBED_SEAT, loadCampaign, type Campaign } from './campaign';

export interface IndexOptions {
  campaignsDir: string;
  campaign: string;
  force?: boolean;
  check?: boolean;
  log?: (line: string) => void;
  /** Injected by tests; production builds one from the campaign's embed seat. */
  embedder?: Embedder;
}

export interface IndexResult {
  indexDir: string;
  rebuilt: boolean;
  /**
   * Whether the index is out of date. Always false after a rebuild. `--check` exists to answer
   * exactly this, so it must be reported, not merely computed.
   */
  stale: boolean;
  meta: IndexMeta | undefined;
  ambiguous: readonly string[];
  /** The lore version the repo is at right now, whether or not the index was rebuilt. */
  loreCommit: string;
}

/** The index is per lore checkout, not per session; `.cache` is git-ignored (v1 design §6.1). */
export function loreIndexDir(campaign: Campaign): string {
  return join(campaign.dir, '.cache', 'lore-index');
}

export function embedderFor(campaign: Campaign): Embedder {
  if (!campaign.embedSeat) {
    throw new Error(
      `Campaign "${campaign.id}" has no embed seat. Add lorePath and seats.embed to campaign.json.`
    );
  }
  const seat = campaign.models[EMBED_SEAT];
  if (!seat) throw new Error(`Campaign "${campaign.id}" has no model for seat "${EMBED_SEAT}".`);
  return new OpenAICompatibleEmbedder(
    { endpoint: seat.endpoint, model: seat.model },
    { timeoutMs: seat.timeoutMs }
  );
}

function requireLorePath(campaign: Campaign): string {
  if (!campaign.lorePath) {
    throw new Error(
      `Campaign "${campaign.id}" has no lorePath. Point it at a cartyx-lore checkout in ` +
        `campaign.json to index lore.`
    );
  }
  return campaign.lorePath;
}

export async function currentIdentity(
  campaign: Campaign,
  embedder: Embedder,
  dims: number | undefined
): Promise<{ identity: IndexIdentity; loreCommit: string }> {
  const lorePath = requireLorePath(campaign);
  const files = await readLoreFiles(lorePath);
  const digest = contentDigest(files);
  return {
    identity: {
      lorePath,
      contentDigest: digest,
      embedModel: embedder.model,
      // The dimension count is only known after an embedding call, so compare against whatever the
      // existing index recorded: a model swap is already caught by `embedModel`.
      dims: dims ?? 0,
      documentPrefix: DOCUMENT_PREFIX,
      queryPrefix: QUERY_PREFIX,
      chunkerVersion: CHUNKER_VERSION,
    },
    loreCommit: await loreVersion(lorePath, digest),
  };
}

/** Builds the index if it is missing or stale; returns the existing one otherwise. */
export async function ensureIndex(options: IndexOptions): Promise<IndexResult> {
  const campaign = await loadCampaign(options.campaignsDir, options.campaign);
  const lorePath = requireLorePath(campaign);
  const embedder = options.embedder ?? embedderFor(campaign);
  const indexDir = loreIndexDir(campaign);
  const existing = await readMeta(indexDir);
  const { identity, loreCommit } = await currentIdentity(campaign, embedder, existing?.dims);
  const stale = options.force === true || isStale(existing, identity);
  if (options.check) {
    // Report the staleness we just computed. Returning a bare `rebuilt: false` here would make
    // `formatIndexSummary` print "Index is fresh" for a stale index — the one question --check
    // exists to answer, answered wrong in the case the operator cares about.
    return {
      indexDir,
      rebuilt: false,
      stale,
      meta: existing,
      ambiguous: existing?.ambiguous ?? [],
      loreCommit,
    };
  }
  if (!stale && existing) {
    return { indexDir, rebuilt: false, meta: existing, ambiguous: [], loreCommit };
  }
  const built = await buildIndex({ lorePath, indexDir, embedder, log: options.log });
  return {
    indexDir,
    rebuilt: true,
    meta: built.meta,
    ambiguous: built.ambiguous,
    loreCommit: built.meta.loreCommit,
  };
}

export async function runIndex(options: IndexOptions): Promise<IndexResult> {
  return ensureIndex(options);
}

export function formatIndexSummary(result: IndexResult): string[] {
  const lines: string[] = [];
  if (!result.meta) {
    lines.push(`No lore index at ${result.indexDir}. Run \`sim index\` to build one.`);
    lines.push(`Lore is at ${result.loreCommit}.`);
    return lines;
  }
  const meta = result.meta;
  lines.push(
    `${result.rebuilt ? 'Built' : 'Index is fresh:'} ${meta.chunks} chunk${
      meta.chunks === 1 ? '' : 's'
    } from ${meta.files} file${meta.files === 1 ? '' : 's'} (${meta.embedModel}, ${meta.dims} dims)`
  );
  lines.push(`Lore commit: ${meta.loreCommit}`);
  lines.push(`Index: ${result.indexDir}`);
  if (result.ambiguous.length > 0) {
    lines.push('');
    lines.push(
      `${result.ambiguous.length} ambiguous entity name${
        result.ambiguous.length === 1 ? '' : 's'
      } fell back to a path id (two Kanka entities share each name):`
    );
    for (const name of result.ambiguous) lines.push(`  ${name}`);
  }
  return lines;
}

/** The lore commit stamped on a session's `session_start`, or undefined if the log has none. */
async function loggedLoreCommit(eventsPath: string): Promise<string | undefined> {
  let text: string;
  try {
    text = await readFile(eventsPath, 'utf8');
  } catch {
    return undefined;
  }
  for (const line of text.split('\n')) {
    if (line.trim() === '') continue;
    try {
      const event = JSON.parse(line) as { type?: string; loreCommit?: string };
      if (event.type === 'session_start') return event.loreCommit;
    } catch {
      // A torn final line is recovered elsewhere; here it simply is not a session_start.
    }
  }
  return undefined;
}

 * `lookup_lore`, a dozen minutes into a session.

export async function benchEmbedSeat(embedder: Embedder): Promise<EmbedBenchResult> {
  const started = Date.now();
  try {
    const vector = await embedder.embedQuery('who teaches applied crystal engines');
    return {
      seat: EMBED_SEAT,
      model: embedder.model,
      reachable: true,
      dims: vector.length,
      latencyMs: Date.now() - started,
      errors: [],
    };
  } catch (error) {
    return {
      seat: EMBED_SEAT,
      model: embedder.model,
      reachable: false,
      dims: null,
      latencyMs: null,
      errors: [error instanceof Error ? error.message : String(error)],
    };
  }
}

export function formatEmbedBench(result: EmbedBenchResult): string[] {
  const lines = [
    `${result.seat}  ${result.model}  ${result.reachable ? 'yes' : 'NO'}  ` +
      `${result.dims === null ? '-' : `${result.dims} dims`}  ` +
      `${result.latencyMs === null ? '-' : `${result.latencyMs} ms`}`,
  ];
  for (const error of result.errors) lines.push(`  ${result.seat}: ${error}`);
  return lines;
}

import { applyEvent, initialState, spokenMinutes, type SimEvent } from '@cartyx-sim/core';

/** Bumped when the audit's shape changes, so stored reports stay comparable across sessions. */
export const AUDIT_VERSION = 1;

export interface GapCluster {
  topic: string;
  count: number;
  queries: { query: string; seq: number }[];
}

export interface InventionEntry {
  seq: number;
  fact: string;
  reason: string;
  query?: string;
  narrationSeq?: number;
  narration?: string;
  /** Where in the spoken session the supported narration falls. */
  spokenMinutes?: number;
}

export interface CoverageEntry {
  entityId: string;
  entityName: string;
  entityType: string;
  lookups: number;
}

export interface NpcEntry {
  npcId: string;
  name: string;
  loreEntityId?: string;
  invented: boolean;
}

export interface SceneEntry {
  seq: number;
  location: string;
  loreEntityId?: string;
}

export interface Contradiction {
  seq: number;
  narration: string;
  chunkIds: string[];
  note: string;
}

export interface Audit {
  auditVersion: number;
  campaign: string;
  session: number;
  loreCommit: string;
  gaps: GapCluster[];
  inventions: InventionEntry[];
  coverage: CoverageEntry[];
  npcs: NpcEntry[];
  scenes: SceneEntry[];
  /** null when the checker pass did not run; the report says why. */
  contradictions: Contradiction[] | null;
}

const STOP_WORDS = new Set([
  'the',
  'a',
  'an',
  'of',
  'is',
  'are',
  'was',
  'were',
  'who',
  'what',
  'where',
  'when',
  'why',
  'how',
  'does',
  'do',
  'did',
  'to',
  'in',
  'on',
  'at',
  'for',
  'and',
  'or',
  'with',
  'about',
]);

export function normalizeQuery(query: string): string[] {
  return query
    .toLowerCase()
    .replace(/[^\p{L}\p{N}\s]+/gu, ' ')
    .split(/\s+/)
    .filter((word) => word.length >= 3 && !STOP_WORDS.has(word));
}

function overlap(a: readonly string[], b: readonly string[]): number {
  const left = new Set(a);
  const shared = new Set(b.filter((word) => left.has(word))).size;
  const smaller = Math.min(left.size, new Set(b).size);
  return smaller === 0 ? 0 : shared / smaller;
}

/**
 * Greedy clustering: a query joins the first cluster whose seed query it shares half or more of
 * the smaller token set with. Good enough to group "the Old Stone Bridge" with "Old Stone Bridge
 * history" without pretending to be topic modelling.
 */
export function clusterGaps(gaps: readonly { query: string; seq: number }[]): GapCluster[] {
  const clusters: { tokens: string[]; entries: { query: string; seq: number }[] }[] = [];
  for (const gap of gaps) {
    const tokens = normalizeQuery(gap.query);
    const existing = clusters.find((cluster) => overlap(cluster.tokens, tokens) >= 0.5);
    if (existing) existing.entries.push(gap);
    else clusters.push({ tokens, entries: [gap] });
  }
  return clusters
    .map((cluster) => ({
      topic: cluster.tokens.join(' '),
      count: cluster.entries.length,
      queries: cluster.entries,
    }))
    .sort((a, b) => b.count - a.count || a.queries[0]!.seq - b.queries[0]!.seq);
}

export function buildAudit(input: {
  campaign: string;
  session: number;
  events: readonly SimEvent[];
}): Omit<Audit, 'contradictions'> {
  const { events } = input;
  const start = events.find((event) => event.type === 'session_start');

  // Fold the log the way the engine does, so the spoken-minute mark on an invention comes from
  // the same `spokenWords` the session clock steered by — not a second word count that could
  // drift from it.
  const narrations = new Map<number, string>();
  const spokenAt = new Map<number, number>();
  let state = initialState();
  for (const event of events) {
    state = applyEvent(state, event);
    if (event.type === 'narration') {
      narrations.set(event.seq, event.text);
      spokenAt.set(event.seq, spokenMinutes(state));
    }
  }

  const gapEntries: { query: string; seq: number }[] = [];
  const coverage = new Map<string, CoverageEntry>();
  for (const event of events) {
    if (event.type !== 'lore_lookup') continue;
    if (event.used.length === 0) {
      gapEntries.push({ query: event.query, seq: event.seq });
      continue;
    }
    // One count per lookup per entity: two chunks of the same NPC are one use of that NPC.
    const seen = new Set<string>();
    for (const hit of event.hits) {
      if (!event.used.includes(hit.chunkId)) continue;
      // A log written before Plan 2B has no entity id; the chunk id is the best key it offers.
      const entityId = hit.entityId ?? hit.chunkId;
      if (seen.has(entityId)) continue;
      seen.add(entityId);
      const existing = coverage.get(entityId);
      if (existing) existing.lookups++;
      else
        coverage.set(entityId, {
          entityId,
          entityName: hit.entityName ?? hit.chunkId,
          // The event records an entity's id and name but not its type, so the audit does not
          // claim one. `sim index`'s meta has types if a richer report ever needs them.
          entityType: 'unknown',
          lookups: 1,
        });
    }
  }

  const inventions: InventionEntry[] = events
    .filter((event) => event.type === 'lore_invention')
    .map((event) => ({
      seq: event.seq,
      fact: event.fact,
      reason: event.reason,
      ...(event.query === undefined ? {} : { query: event.query }),
      ...(event.narrationSeq === undefined ? {} : { narrationSeq: event.narrationSeq }),
      ...(event.narrationSeq !== undefined && narrations.has(event.narrationSeq)
        ? {
            narration: narrations.get(event.narrationSeq)!,
            spokenMinutes: spokenAt.get(event.narrationSeq)!,
          }
        : {}),
    }));

  const npcs: NpcEntry[] = events
    .filter((event) => event.type === 'npc_introduced')
    .map((event) => ({
      npcId: event.npcId,
      name: event.name,
      ...(event.loreEntityId === undefined ? {} : { loreEntityId: event.loreEntityId }),
      // The DM's own answer, not a guess. `invented` is required on npc_introduced and is
      // model-supplied independently of loreEntityId (tools/narrative.ts passes args.invented
      // through; the tool says "set loreEntityId when the NPC comes from the lore; set invented
      // to true if you made them up"). Deriving it from linkage discards the answer we asked for.
      invented: event.invented,
    }));

  const scenes: SceneEntry[] = events
    .filter((event) => event.type === 'scene_change')
    .map((event) => ({
      seq: event.seq,
      location: event.location,
      ...(event.loreEntityId === undefined ? {} : { loreEntityId: event.loreEntityId }),
    }));

  return {
    auditVersion: AUDIT_VERSION,
    campaign: input.campaign,
    session: input.session,
    loreCommit: start?.loreCommit ?? 'unknown',
    gaps: clusterGaps(gapEntries),
    inventions,
    coverage: [...coverage.values()].sort((a, b) => b.lookups - a.lookups),
    npcs,
    scenes,
  };
}

export function renderAuditMarkdown(audit: Audit): string {
  const lines: string[] = [
    `# Lore audit — ${audit.campaign} session ${audit.session}`,
    '',
    `**Lore commit:** ${audit.loreCommit}`,
    '',
    '## Gaps',
    '',
  ];
  if (audit.gaps.length === 0) lines.push('No lookup came back empty.', '');
  for (const cluster of audit.gaps) {
    lines.push(`### ${cluster.topic || '(no keywords)'} — ${cluster.count}`, '');
    for (const entry of cluster.queries) lines.push(`- \`${entry.query}\` (seq ${entry.seq})`);
    lines.push('');
  }

  lines.push('## Inventions', '');
  if (audit.inventions.length === 0) lines.push('Nothing was invented.', '');
  for (const invention of audit.inventions) {
    lines.push(`### seq ${invention.seq}`, '');
    lines.push(`**Fact:** ${invention.fact}`, '');
    lines.push(`**Why:** ${invention.reason}`, '');
    if (invention.query) lines.push(`**After looking up:** \`${invention.query}\``, '');
    if (invention.narration) {
      const at =
        invention.spokenMinutes === undefined
          ? ''
          : ` at ${invention.spokenMinutes.toFixed(1)} spoken min`;
      lines.push(`**Supports (seq ${invention.narrationSeq}${at}):** ${invention.narration}`, '');
    } else {
      lines.push('**Supports:** no narration in that turn.', '');
    }
  }

  lines.push('## Coverage', '');
  if (audit.coverage.length === 0) lines.push('No lore was used.', '');
  else {
    lines.push('| entity | id | lookups |', '| --- | --- | --- |');
    for (const entry of audit.coverage) {
      lines.push(`| ${entry.entityName} | \`${entry.entityId}\` | ${entry.lookups} |`);
    }
    lines.push('');
  }
  lines.push(
    `NPCs introduced: ${audit.npcs.length} (${audit.npcs.filter((npc) => npc.invented).length} invented).`,
    '',
    `Scenes: ${audit.scenes.length} (${audit.scenes.filter((scene) => scene.loreEntityId).length} tied to lore).`,
    ''
  );

  lines.push('## Possible contradictions', '');
  if (audit.contradictions === null) {
    lines.push('Skipped: no checker seat, no lore index, or `--no-model` was passed.', '');
  } else if (audit.contradictions.length === 0) {
    lines.push('The checker found none.', '');
  } else {
    for (const finding of audit.contradictions) {
      lines.push(`- **seq ${finding.seq}** — ${finding.note}`);
      lines.push(`  > ${finding.narration}`);
      lines.push(`  Chunks: ${finding.chunkIds.map((id) => `\`${id}\``).join(', ')}`);
    }
    lines.push('');
  }
  return `${lines.join('\n')}\n`;
}

```ts
/** Bump when the prompt changes, so a stored audit says which prompt produced it. */
export const CONTRADICTION_PROMPT_VERSION = 1;

const CONTRADICTION_SYSTEM =
  "You compare a game master's narration against the world lore they looked up. " +
  'Answer only with JSON: {"conflicts": boolean, "note": string}. ' +
  'Set conflicts to true only when the narration states something the lore contradicts — a ' +
  'different name, role, place, or fact. Detail the lore simply does not mention is NOT a ' +
  'conflict; the game master is allowed to add to the world. Keep the note to one sentence.';

/**
 * One pass over the narrations that followed a lore lookup with usable hits. A narration after a
 * gap has nothing to contradict, so it is not asked about — which also keeps the pass to a handful
 * of calls a session.
 */
export async function findContradictions(input: {
  events: readonly SimEvent[];
  model: ModelClient;
  seat: string;
  chunkText: (chunkId: string) => Promise<string | undefined>;
}): Promise<Contradiction[]> {
  const found: Contradiction[] = [];
  let pending: string[] | undefined;
  for (const event of input.events) {
    if (event.type === 'lore_lookup') {
      const used = event.hits.filter((hit) => event.used.includes(hit.chunkId));
      pending = used.length === 0 ? undefined : used.map((hit) => hit.chunkId);
      continue;
    }
    if (event.type !== 'narration' || !pending) continue;
    const chunkIds = pending;
    // Only the first narration after a lookup is checked: later ones have drifted away from what
    // was retrieved, and asking about each would multiply the calls for little signal.
    pending = undefined;

    // The log records chunk ids and scores, never the text. Bare ids tell the checker nothing, so
    // the chunks are re-fetched from the index; a chunk the index no longer holds (the lore moved
    // on since the session) means there is nothing to compare, and the narration is skipped.
    const texts: string[] = [];
    for (const chunkId of chunkIds) {
      const text = await input.chunkText(chunkId);
      if (text !== undefined) texts.push(`[${chunkId}] ${text}`);
    }
    if (texts.length === 0) continue;

    const reply = await input.model.complete({
      seat: input.seat,
      messages: [
        { role: 'system', content: CONTRADICTION_SYSTEM },
        {
          role: 'user',
          content: `Lore the game master retrieved:\n${texts.join('\n\n')}\n\nNarration:\n${event.text}`,
        },
      ],
      tools: [],
    });
    const parsed = parseJsonReply<{ conflicts?: unknown; note?: unknown }>(reply.text);
    if (!parsed || parsed.conflicts !== true) continue;
    found.push({
      seq: event.seq,
      narration: event.text,
      chunkIds,
      note: typeof parsed.note === 'string' ? parsed.note : '',
    });
  }
  return found;
}

**Caveat on 0.69 — re-tune if real sessions report false gaps.** The margin is tight: only ~0.05
separates the lowest observed correct answer (0.740) from the highest observed miss (0.638), and the
sample behind it is five clean, well-formed queries against a corpus that happens to answer four of
them directly. Real sessions will ask messier, more oblique questions than these, and those will
plausibly score lower even when the lore does answer them. 0.69 errs in the safer direction — too
high produces false gaps, which `sim audit` makes visible and a lore author can act on, whereas too
low produces invisible pollution (irrelevant lore accepted as relevant, silently) — but if a real
session's audit reports gaps for lore that demonstrably exists in the corpus, that is the signal to
lower this threshold, with the numbers above as the baseline to re-tune from.

4 (`which professor coaches technical dueling?`) returns the right professor (Wenda Gale) at 0.740

correct answer — so `loreThreshold` functions as a gap detector (is there _anything_ topically
relevant at all) and not a relevance ranker (which specific hit is the right one). Anyone re-tuning
this threshold, or building on top of it, should keep that distinction in mind.
